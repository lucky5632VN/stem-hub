### CHI TIẾT CÁC HÀM TÍNH TOÁN ĐỘC LẬP BỔ SUNG

```python
# 1. HÓA HỌC: SUPP_CHEM_01 (Tính thể tích khí H2 đkc giải phóng từ 5.4g Al)
def calc_supp_chem_01(m_Al=5.4):
    M_Al = 27.0
    n_Al = m_Al / M_Al # 0.2 mol
    n_H2 = n_Al * 1.5   # 2Al + 6HCl -> 2AlCl3 + 3H2
    V_H2 = n_H2 * 24.79 # Điều kiện chuẩn 25 độ C, 1 bar: 7.437 lít
    return round(V_H2, 3) # 7.437

# 2. HÓA HỌC: SUPP_CHEM_02 (Tính pH dung dịch NaOH 0.01M)
def calc_supp_chem_02(c_NaOH=0.01):
    import math
    pOH = -math.log10(c_NaOH) # 2.0
    pH = 14.0 - pOH          # 12.0
    return int(round(pH))     # 12

# 3. SINH HỌC: SUPP_BIO_01 (Tính số liên kết hiđro của ADN 3000 nu, 20% A)
def calc_supp_bio_01(N=3000, pct_A=0.20):
    A = int(N * pct_A)       # 600
    G = int(N / 2 - A)       # 900
    H = 2 * A + 3 * G        # 2*600 + 3*900 = 3900
    return H                 # 3900

# 4. SINH HỌC: SUPP_BIO_02 (Số loại giao tử của cá thể AaBbDdEe)
def calc_supp_bio_02(n_hetero=4):
    return 2 ** n_hetero     # 2^4 = 16

# 5. VẬT LÝ: SUPP_PHYS_01 (Tính bước sóng sóng cơ v=12 m/s, f=60 Hz)
def calc_supp_phys_01(v=12.0, f=60.0):
    lam = v / f              # 12 / 60 = 0.2 m = 20 cm
    return round(lam, 2)     # 0.2 m (20 cm)

# 6. VẬT LÝ: SUPP_PHYS_02 (Nhiệt nóng chảy 0.5 kg nước đá, lambda = 3.34e5 J/kg)
def calc_supp_phys_02(m=0.5, lam=3.34e5):
    Q = m * lam              # 0.5 * 334000 = 167000 J = 167 kJ
    return int(Q)            # 167000 (167 kJ)

# 7. TOÁN HỌC: SUPP_MATH_01 (Tích phân I = int_1^2 (2x + 1) dx)
def calc_supp_math_01():
    # F(x) = x^2 + x => F(2) - F(1) = (4 + 2) - (1 + 1) = 6 - 2 = 4
    return (2**2 + 2) - (1**2 + 1) # 4
```
